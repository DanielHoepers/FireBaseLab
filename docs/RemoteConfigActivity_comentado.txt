public class RemoteConfigActivity extends AppCompatActivity { // L18: Declara a tela como uma Activity Android.

    private TextView origem; // L26: Guarda o campo que informa a origem dos valores.

    private com.google.android.material.button.MaterialButton buscar; // L28: Guarda o botão que inicia uma nova busca.

    @Override // L30: Indica que onCreate substitui um método da classe-base.
    protected void onCreate(Bundle savedInstanceState) { // L31: Começa o método chamado quando a tela é criada.
        super.onCreate(savedInstanceState); // L32: Executa primeiro a inicialização padrão da Activity.
        setContentView(R.layout.remote_config_demo); // L33: Carrega o layout visual desta tela.

        findViewById(R.id.btnVoltar).setOnClickListener(v -> finish()); // L35: Faz o botão Voltar encerrar esta Activity.
        titulo = findViewById(R.id.textTituloConfig); // L36: Encontra o campo do título no layout.
        mensagem = findViewById(R.id.textMensagemConfig); // L37: Encontra o campo da mensagem no layout.
        origem = findViewById(R.id.textOrigemConfig); // L38: Encontra o campo que mostra a origem dos valores.

                .setMinimumFetchIntervalInSeconds(intervalo) // L46: Aplica às opções o intervalo mínimo definido acima.
                .build(); // L47: Finaliza a criação das opções de busca.

        Tasks.whenAll(remoteConfig.setConfigSettingsAsync(settings), // L49: Inicia as duas tarefas assíncronas e espera ambas terminarem.

                .addOnCompleteListener(this, task -> { // L51: Define o que fazer quando as duas tarefas terminarem.

                        Toast.makeText(this, "Falha ao iniciar Remote Config.", Toast.LENGTH_LONG).show(); // L54: Exibe uma mensagem temporária de erro.
                        return; // L55: Interrompe este retorno antecipadamente em caso de falha.

                }); // L59: Fecha a função chamada quando a preparação termina.

    private void buscarConfiguracoes() { // L62: Começa o método executado ao tocar em Buscar configurações.
        buscar.setEnabled(false); // L63: Desabilita o botão enquanto a busca está em andamento.
        origem.setText("Buscando configurações no Firebase..."); // L64: Informa ao usuário que a busca começou.

        remoteConfig.fetchAndActivate().addOnCompleteListener(this, task -> { // L66: Busca e ativa valores, depois registra uma função para o resultado.
            buscar.setEnabled(true); // L67: Habilita novamente o botão quando a operação termina.
            if (!task.isSuccessful()) { // L68: Verifica se a busca falhou.
                origem.setText("Falha na busca. Os valores atuais continuam visíveis."); // L69: Mostra que os valores já exibidos permanecem na tela.

                        + (task.getException() == null ? "verifique a conexão" // L71: Usa uma mensagem genérica se a exceção não estiver disponível.
                        : task.getException().getMessage()), Toast.LENGTH_LONG).show(); // L72: Caso exista exceção, mostra sua mensagem e finaliza o Toast.
                return; // L73: Sai da função sem aplicar novos valores após uma falha.
            } // L74: Fecha o bloco de tratamento da falha.

            boolean corValida = aplicarValores(); // L76: Atualiza a tela e registra se a cor recebida era válida.
            if (!corValida) { // L77: Verifica se a cor recebida era inválida.
                Toast.makeText(this, "cor_destaque inválida; usando a cor padrão.", // L78: Cria o aviso de que a cor padrão foi utilizada.

                        Toast.LENGTH_SHORT).show(); // L82: Exibe a mensagem de sucesso por um período curto.
            } // L83: Fecha a decisão entre aviso de cor e sucesso.
        }); // L84: Fecha a função executada após a busca.
    } // L85: Fecha o método buscarConfiguracoes.

    private boolean aplicarValores() { // L87: Começa o método que atualiza os elementos visuais.
        titulo.setText(remoteConfig.getString(TITULO_APP)); // L88: Lê titulo_app e coloca o resultado no título.
        mensagem.setText(remoteConfig.getString(MENSAGEM_APP)); // L89: Lê mensagem_app e coloca o resultado na mensagem.

        int cor; // L91: Declara a variável que guardará a cor calculada.
        boolean corValida = true; // L92: Considera a cor válida até que uma conversão falhe.
        try { // L93: Inicia uma tentativa de interpretar a cor recebida.
            cor = Color.parseColor(remoteConfig.getString(COR_DESTAQUE).trim()); // L94: Lê cor_destaque, remove espaços e converte o texto em cor.
        } catch (IllegalArgumentException exception) { // L95: Captura o erro causado por um valor de cor inválido.
            cor = getColor(R.color.firebase_blue); // L96: Usa o azul local como alternativa.
            corValida = false; // L97: Registra que a cor recebida era inválida.
        } // L98: Fecha o tratamento da conversão da cor.
        titulo.setTextColor(cor); // L99: Aplica a cor ao texto do título.
        faixaDestaque.setBackgroundColor(cor); // L100: Aplica a cor à faixa horizontal.
        buscar.setBackgroundTintList(ColorStateList.valueOf(cor)); // L101: Aplica a cor ao fundo do botão.

        boolean remoto = remoteConfig.getValue(TITULO_APP).getSource() // L103: Consulta a origem do valor usado para o título.
                == FirebaseRemoteConfig.VALUE_SOURCE_REMOTE; // L104: Confere se o título veio de uma configuração remota ativada.
        origem.setText(remoto ? "Exibindo valores publicados no Firebase" // L105: Escolhe o texto que informa a origem dos valores.
                : "Exibindo valores padrão do aplicativo"); // L106: Usa o texto de valores locais quando não há título remoto.
        return corValida; // L107: Devolve true para cor válida ou false para cor inválida.
    } // L108: Fecha o método aplicarValores.
} // L109: Fecha a classe RemoteConfigActivity.
